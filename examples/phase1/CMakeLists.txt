cmake_minimum_required(VERSION 3.16)
project(amr_phase1 LANGUAGES CXX)
add_executable(kinematics_demo main.cpp)
target_compile_features(kinematics_demo PRIVATE cxx_std_17)
target_compile_options(kinematics_demo PRIVATE -Wall -Wextra -Wpedantic)

  add_executable(kinematics_test test_kinematics.cpp)
  target_compile_features(kinematics_test PRIVATE cxx_std_17)
  add_test(NAME kinematics COMMAND kinematics_test)
endif()
